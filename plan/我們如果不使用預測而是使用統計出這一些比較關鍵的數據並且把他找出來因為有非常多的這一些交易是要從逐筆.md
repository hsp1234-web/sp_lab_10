## 核心理念：從訂單流拆解市場行為

### 為什麼1分鐘K線不夠？

```python
# 1分鐘K線只能看到：
{
    'open': 580.0,
    'high': 582.0,
    'low': 579.0,
    'close': 581.0,
```

## 階段1：逐筆交易特徵提取（純統計，不用AI）

### 關鍵統計特徵

```python
import pandas as pd
import numpy as np
from datetime import datetime, timedelta

class TickDataAnalyzer:
    """逐筆交易分析器（純統計方法）"""
    
    def __init__(self):
        self.finmind_api = FinMindAPI()
    
    def extract_tick_features(self, symbol, date):
        """從逐筆交易提取統計特徵"""
        
        # 取得當日逐筆交易（FinMind付費功能）
        ticks = self.finmind_api.get_tick_data(symbol, date)
        
        # ==== 特徵1：大單統計 ====
        features = {}
        
        # 定義大單閾值（根據該股日均量動態調整）
        avg_tick_size = ticks['volume'].median()
        large_order_threshold = avg_tick_size * 5  # 5倍中位數
        
        large_orders = ticks[ticks['volume'] >= large_order_threshold]
        
        features['large_order_count'] = len(large_orders)
        features['large_order_buy_count'] = len(large_orders[large_orders['side'] == 'buy'])
        features['large_order_sell_count'] = len(large_orders[large_orders['side'] == 'sell'])
        features['large_order_imbalance'] = (
            features['large_order_buy_count'] - features['large_order_sell_count']
        ) / (features['large_order_count'] + 1)
        
        # ==== 特徵2：主動買賣壓力 ====
        # 主動買單（以賣價成交）vs 主動賣單（以買價成交）
        aggressive_buys = ticks[ticks['tick_type'] == 'uptick']
        aggressive_sells = ticks[ticks['tick_type'] == 'downtick']
        
        features['aggressive_buy_volume'] = aggressive_buys['volume'].sum()
        features['aggressive_sell_volume'] = aggressive_sells['volume'].sum()
        features['buy_sell_pressure'] = (
            features['aggressive_buy_volume'] / 
            (features['aggressive_sell_volume'] + features['aggressive_buy_volume'])
        )
        
        # ==== 特徵3：時段集中度 ====
        # 開盤、盤中、尾盤的交易分布
        ticks['time_slot'] = pd.cut(
            ticks['time'].dt.hour * 60 + ticks['time'].dt.minute,
            bins=[0, 540, 780, 805, 1440],  # 9:00前, 9:00-13:00, 13:00-13:25, 收盤後
            labels=['pre_open', 'morning', 'afternoon', 'close']
        )
        
        slot_volumes = ticks.groupby('time_slot')['volume'].sum()
        features['open_30min_volume_pct'] = (
            ticks[ticks['time'].dt.hour == 9]['volume'].sum() / 
            ticks['volume'].sum()
        )
        features['close_30min_volume_pct'] = (
            ticks[ticks['time'].dt.hour >= 13]['volume'].sum() / 
            ticks['volume'].sum()
        )
        
        # ==== 特徵4：價格衝擊力 ====
        # 單位成交量造成的價格變動
        ticks['price_change'] = ticks['price'].diff()
        ticks['volume_bucket'] = pd.qcut(ticks['volume'], q=10, labels=False)
        
        price_impact = ticks.groupby('volume_bucket').agg({
            'price_change': 'mean',
            'volume': 'mean'
        })
        
        # 大單的價格衝擊力（檢測是否有主力護盤/打壓）
        features['large_order_price_impact'] = (
            large_orders['price'].diff().mean() / 
            large_orders['volume'].mean()
        )
        
        # ==== 特徵5：訂單簿失衡 ====
        # 每5秒的委買委賣量差距
        five_sec_data = self.finmind_api.get_5sec_data(symbol, date)
        
        five_sec_data['order_imbalance'] = (
            (five_sec_data['bid_volume'] - five_sec_data['ask_volume']) /
            (five_sec_data['bid_volume'] + five_sec_data['ask_volume'])
        )
        
        features['avg_order_imbalance'] = five_sec_data['order_imbalance'].mean()
        features['max_order_imbalance'] = five_sec_data['order_imbalance'].max()
        features['order_imbalance_volatility'] = five_sec_data['order_imbalance'].std()
        
        # ==== 特徵6：成交密集區 ====
        # VWAP (Volume Weighted Average Price) 與當前價的關係
        features['vwap'] = (ticks['price'] * ticks['volume']).sum() / ticks['volume'].sum()
        features['price_vs_vwap'] = (ticks['price'].iloc[-1] / features['vwap']) - 1
        
        # 成交量加權標準差（價格離散度）
        features['vwap_std'] = np.sqrt(
            ((ticks['price'] - features['vwap'])**2 * ticks['volume']).sum() /
            ticks['volume'].sum()
        )
        
        # ==== 特徵7：異常交易偵測 ====
        # 識別突發大量交易
        ticks['rolling_volume'] = ticks['volume'].rolling(window=100, min_periods=1).mean()
        ticks['volume_spike'] = ticks['volume'] / (ticks['rolling_volume'] + 1)
        
        spike_events = ticks[ticks['volume_spike'] > 3]  # 超過平均3倍
        
        features['spike_count'] = len(spike_events)
        features['spike_total_volume'] = spike_events['volume'].sum()
        features['spike_volume_pct'] = (
            features['spike_total_volume'] / ticks['volume'].sum()
        )
        
        # 檢測spike時的價格方向
        features['spike_price_direction'] = np.sign(
            spike_events['price'].diff().sum()
        )
        
        # ==== 特徵8：連續性分析 ====
        # 連續大單買入/賣出（可能是掃單）
        ticks['is_large'] = ticks['volume'] >= large_order_threshold
        ticks['large_streak'] = (
            ticks['is_large'] != ticks['is_large'].shift()
        ).cumsum()
        
        streak_lengths = ticks[ticks['is_large']].groupby('large_streak').size()
        features['max_large_order_streak'] = streak_lengths.max() if len(streak_lengths) > 0 else 0
        
        # ==== 特徵9：掛單深度變化 ====
        # 從5秒數據分析委買委賣掛單變化
        five_sec_data['bid_depth_change'] = five_sec_data['bid_volume'].diff()
        five_sec_data['ask_depth_change'] = five_sec_data['ask_volume'].diff()
        
        features['avg_bid_depth_increase'] = five_sec_data[
            five_sec_data['bid_depth_change'] > 0
        ]['bid_depth_change'].mean()
        
        features['avg_ask_depth_increase'] = five_sec_data[
            five_sec_data['ask_depth_change'] > 0
        ]['ask_depth_change'].mean()
        
        # ==== 特徵10：尾盤拉抬/殺尾 ====
        last_10_min = ticks[ticks['time'] >= '13:20:00']
        
        if len(last_10_min) > 0:
            features['close_price_change'] = (
                last_10_min['price'].iloc[-1] / last_10_min['price'].iloc[^0] - 1
            )
            features['close_volume_pct'] = (
                last_10_min['volume'].sum() / ticks['volume'].sum()
            )
            features['close_large_order_count'] = len(
                last_10_min[last_10_min['volume'] >= large_order_threshold]
            )
        
        return features
```


## 階段2：用AI「解讀」統計特徵（不是預測）

### 核心：讓AI回答「發生了什麼」而非「會發生什麼」

```python
class MicrostructureInterpreter:
    """市場微觀結構解讀器（用小模型解讀統計結果）"""
    
    def __init__(self):
        self.model_270m = "gemma3:270m"  # 簡單分類
        self.model_4b = "gemma3:4b"      # 深度解讀
    
    def interpret_tick_features(self, symbol, date, features):
        """逐步解讀統計特徵（多次提問）"""
        
        interpretation = {}
        
        # ==== 問題1：大單行為判斷 ====
        prompt_1 = f"""根據統計數據，判斷大單交易行為。

數據：
- 大單總數: {features['large_order_count']}
- 大單買入: {features['large_order_buy_count']}
- 大單賣出: {features['large_order_sell_count']}
- 買賣不平衡度: {features['large_order_imbalance']:.2f} (-1到1)

只回答：主力買超 / 主力賣超 / 多空均衡 / 無明顯大單"""
        
        interpretation['large_order_behavior'] = self.ask_model(prompt_1, self.model_270m)
        
        # ==== 問題2：主動買賣壓力判斷（帶問題1結果）====
        prompt_2 = f"""大單行為「{interpretation['large_order_behavior']}」。

主動買賣數據：
- 主動買盤: {features['aggressive_buy_volume']:,} 張
- 主動賣盤: {features['aggressive_sell_volume']:,} 張
- 買賣壓力比: {features['buy_sell_pressure']:.2%}

判斷市場情緒（只回答）：強烈買盤 / 買盤占優 / 中性 / 賣壓占優 / 強烈賣壓"""
        
        interpretation['market_sentiment'] = self.ask_model(prompt_2, self.model_270m)
        
        # ==== 問題3：時段交易模式分析 ====
        prompt_3 = f"""分析交易時段分布。

時段數據：
- 開盤30分鐘成交占比: {features['open_30min_volume_pct']:.1%}
- 尾盤30分鐘成交占比: {features['close_30min_volume_pct']:.1%}
- 尾盤價格變化: {features.get('close_price_change', 0):.2%}
- 尾盤大單: {features.get('close_large_order_count', 0)} 筆

判斷交易模式（只回答）：開盤搶進 / 尾盤拉抬 / 尾盤殺尾 / 平穩分布 / 盤中震盪"""
        
        interpretation['trading_pattern'] = self.ask_model(prompt_3, self.model_4b)
        
        # ==== 問題4：異常交易事件解讀 ====
        prompt_4 = f"""解讀異常交易事件。

異常數據：
- 突發大量次數: {features['spike_count']}
- 突發量占總量: {features['spike_volume_pct']:.1%}
- 突發時價格方向: {'+' if features['spike_price_direction'] > 0 else '-' if features['spike_price_direction'] < 0 else '0'}
- 連續大單最長: {features['max_large_order_streak']} 筆

判斷異常原因（只回答）：突發利多 / 掃單搶貨 / 突發利空 / 倒貨出場 / 正常交易"""
        
        interpretation['anomaly_reason'] = self.ask_model(prompt_4, self.model_4b)
        
        # ==== 問題5：訂單簿失衡分析 ====
        prompt_5 = f"""分析委買委賣掛單狀況。

訂單簿數據：
- 平均委買委賣不平衡: {features['avg_order_imbalance']:.2f}
- 最大不平衡度: {features['max_order_imbalance']:.2f}
- 不平衡波動率: {features['order_imbalance_volatility']:.2f}

判斷掛單狀況（只回答）：買單堆積 / 賣單堆積 / 頻繁換單 / 穩定掛單 / 異常抽單"""
        
        interpretation['orderbook_status'] = self.ask_model(prompt_5, self.model_4b)
        
        # ==== 問題6：價格衝擊力解讀 ====
        prompt_6 = f"""分析大單對價格的影響力。

價格衝擊數據：
- 大單平均價格衝擊: {features['large_order_price_impact']:.4f}
- VWAP vs 收盤價: {features['price_vs_vwap']:.2%}
- 價格離散度: {features['vwap_std']:.2f}

判斷價格操控（只回答）：主力護盤 / 主力打壓 / 自然波動 / 價格僵固 / 劇烈震盪"""
        
        interpretation['price_control'] = self.ask_model(prompt_6, self.model_4b)
        
        # ==== 問題7：綜合行為推論（帶所有前置答案）====
        prompt_7 = f"""綜合以上分析，推論今日主力行為。

統計摘要：
- 大單: {interpretation['large_order_behavior']}
- 情緒: {interpretation['market_sentiment']}
- 模式: {interpretation['trading_pattern']}
- 異常: {interpretation['anomaly_reason']}
- 掛單: {interpretation['orderbook_status']}
- 價控: {interpretation['price_control']}

推論主力意圖（用一句話，不超過30字）："""
        
        interpretation['inferred_intention'] = self.ask_model(prompt_7, self.model_4b)
        
        # ==== 問題8：是否值得跟隨 ====
        prompt_8 = f"""基於今日分析，判斷是否有跟隨價值。

主力意圖: {interpretation['inferred_intention']}
大單行為: {interpretation['large_order_behavior']}
市場情緒: {interpretation['market_sentiment']}

是否建議跟隨（只回答）：強烈跟隨 / 謹慎跟隨 / 觀望 / 不建議 / 反向操作"""
        
        interpretation['follow_recommendation'] = self.ask_model(prompt_8, self.model_4b)
        
        return interpretation
```

## 階段4：累積學習與模式識別

```python
class PatternMemorySystem:
    """模式記憶系統（累積對盤面的認知）"""
    
    def __init__(self):
        self.db = sqlite3.connect('trading_microstructure.db')
        self.model = "gemma3:4b"
    
    def identify_pattern(self, current_features):
        """識別當前特徵是否匹配已知模式"""
        
        # 從資料庫讀取歷史相似模式
        query = """
        SELECT pattern_id, pattern_name, feature_signature,
               avg_next_1d_return, win_rate
        FROM pattern_memory
        WHERE occurrence_count >= 5  -- 至少出現過5次
        ORDER BY win_rate DESC
        """
        
        known_patterns = pd.read_sql(query, self.db)
        
        matches = []
        for _, pattern in known_patterns.iterrows():
            signature = json.loads(pattern['feature_signature'])
            similarity = self.calculate_similarity(current_features, signature)
            
            if similarity > 0.8:  # 高度相似
                matches.append({
                    'pattern_name': pattern['pattern_name'],
                    'similarity': similarity,
                    'expected_return': pattern['avg_next_1d_return'],
                    'win_rate': pattern['win_rate']
                })
        
        return matches
    
    def calculate_similarity(self, features1, features2):
        """計算特徵相似度"""
        # 使用餘弦相似度或歐式距離
        # 只比較關鍵特徵
        key_features = [
            'large_order_imbalance',
            'buy_sell_pressure',
            'spike_volume_pct',
            'avg_order_imbalance'
        ]
        
        vec1 = np.array([features1.get(k, 0) for k in key_features])
        vec2 = np.array([features2.get(k, 0) for k in key_features])
        
        return np.dot(vec1, vec2) / (np.linalg.norm(vec1) * np.linalg.norm(vec2))
    
    def learn_new_pattern(self, date, symbol, features, interpretation, actual_return):
        """從新數據中學習模式"""
        
        # ==== AI判斷是否值得記憶 ====
        prompt = f"""評估這個交易模式是否值得記憶。

特徵：
- 大單不平衡: {features['large_order_imbalance']:.2f}
- 買賣壓力: {features['buy_sell_pressure']:.2%}
- 異常交易: {features['spike_count']} 次
- AI解讀: {interpretation['inferred_intention']}

實際結果：
- 隔日報酬: {actual_return:.2%}

這個模式是否有記憶價值？只回答：高價值 / 中等價值 / 低價值 / 無價值"""
        
        memory_value = self.ask_model(prompt)
        
        if memory_value in ['高價值', '中等價值']:
            # 生成模式名稱
            pattern_name = f"{interpretation['large_order_behavior']}_{interpretation['trading_pattern']}_{date}"
            
            # 儲存模式
            cursor = self.db.cursor()
            cursor.execute("""
            INSERT INTO pattern_memory
            (pattern_name, feature_signature, avg_next_1d_return, win_rate)
            VALUES (?, ?, ?, ?)
            """, (
                pattern_name,
                json.dumps(features),
                actual_return,
                1.0 if actual_return > 0 else 0.0
            ))
            self.db.commit()
    
    def update_pattern_performance(self, pattern_id, new_return):
        """更新模式績效（移動平均）"""
        cursor = self.db.cursor()
        
        # 取得舊數據
        cursor.execute("""
        SELECT occurrence_count, avg_next_1d_return, win_rate
        FROM pattern_memory WHERE pattern_id = ?
        """, (pattern_id,))
        
        old_count, old_avg_return, old_win_rate = cursor.fetchone()
        
        # 更新統計
        new_count = old_count + 1
        new_avg_return = (old_avg_return * old_count + new_return) / new_count
        new_win_rate = (old_win_rate * old_count + (1 if new_return > 0 else 0)) / new_count
        
        cursor.execute("""
        UPDATE pattern_memory
        SET occurrence_count = ?,
            avg_next_1d_return = ?,
            win_rate = ?,
            last_seen_date = ?,
            updated_at = CURRENT_TIMESTAMP
        WHERE pattern_id = ?
        """, (new_count, new_avg_return, new_win_rate, datetime.now().date(), pattern_id))
        
        self.db.commit()
```

```python
def backtest_microstructure_strategy(symbol, start_date, end_date):
    """回測基於微觀結構的策略"""
    
    trading_days = get_trading_days(start_date, end_date)
    results = []
    
    for date in tqdm(trading_days, desc="回測進度"):
        # 分析當日
        rec = daily_microstructure_analysis(symbol, date)

        if rec['action'] == 'BUY':
            entry_price = get_close_price(symbol, date)
            exit_price = get_close_price(symbol, get_next_trading_day(date))
            actual_return = (exit_price / entry_price) - 1
            
            results.append({
```
