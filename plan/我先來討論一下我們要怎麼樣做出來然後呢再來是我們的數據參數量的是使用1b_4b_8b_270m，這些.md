from concurrent.futures import ThreadPoolExecutor

class FastPreProcessor:
    """使用超小模型做預處理"""
    
    def __init__(self):
        self.model = "gemma3:270m"  # 極快的小模型
        
    def batch_classify_news(self, news_list):
        """批量分類新聞情緒（每秒可處理30-50條）"""
        prompts = []
        for news in news_list:
            prompts.append(f"""新聞標題：{news['title'][:100]}

情緒分類（只回答數字）：
1=利多 2=利空 3=中性 4=不確定

回答：""")
        
        # 並行處理
        with ThreadPoolExecutor(max_workers=5) as executor:
            results = list(executor.map(
                lambda p: self.ask_tiny_model(p), 
                prompts
            ))
        
        return results
    
    def quick_anomaly_detection(self, daily_data):
        """快速識別異常數據點"""
        prompt = f"""檢查數據是否異常：
成交量: {daily_data['volume']} (平均: {daily_data['avg_volume']})
漲跌幅: {daily_data['return']:.2%}
振幅: {daily_data['amplitude']:.2%}

是否異常？只回答：是 / 否"""
        
        return self.ask_tiny_model(prompt)
    
    def categorize_trading_pattern(self, price_pattern):
        """快速分類價格型態"""
        prompt = f"""價格型態分類（只回答數字）：
近5日: {price_pattern['last_5d']}

1=突破 2=回檔 3=整理 4=破底 5=不明

回答："""
        
        return int(self.ask_tiny_model(prompt) or "5")

```python
class TechnicalAnalyzer:
    """技術面分析（科學運算+AI驗證）"""
    
    def calculate_all_indicators(self, symbol, days=200):
        """預先計算所有技術指標（純Python，不用AI）"""
        import pandas as pd
        import talib
        
        # 從FinMind取得還原股價
        df = self.get_adjusted_price(symbol, days)
        
        # 趨勢指標
        df['SMA_20'] = talib.SMA(df['close'], 20)
        df['SMA_60'] = talib.SMA(df['close'], 60)
        df['SMA_240'] = talib.SMA(df['close'], 240)
        
        # 動能指標
        df['RSI_14'] = talib.RSI(df['close'], 14)
        df['MACD'], df['MACD_signal'], df['MACD_hist'] = talib.MACD(df['close'])
        df['CCI'] = talib.CCI(df['high'], df['low'], df['close'], 14)
        
        # 波動指標
        df['ATR'] = talib.ATR(df['high'], df['low'], df['close'], 14)
        df['BB_upper'], df['BB_middle'], df['BB_lower'] = talib.BBANDS(df['close'])
        
        # 成交量指標
        df['OBV'] = talib.OBV(df['close'], df['volume'])
        df['Volume_MA20'] = talib.SMA(df['volume'], 20)
        
        # 型態識別（Talib內建100+種型態）
        df['DOJI'] = talib.CDLDOJI(df['open'], df['high'], df['low'], df['close'])
        df['HAMMER'] = talib.CDLHAMMER(df['open'], df['high'], df['low'], df['close'])
        df['ENGULFING'] = talib.CDLENGULFING(df['open'], df['high'], df['low'], df['close'])
        
        return df
    
    def ai_interpret_technical(self, df, date):
        """AI解讀技術指標（只問關鍵問題）"""
        
        latest = df.loc[date]
        
        # ==== 問題1：趨勢方向確認 ====
        prompt_1 = f"""判斷趨勢方向。

均線排列：
- MA20: {latest['SMA_20']:.2f}
- MA60: {latest['SMA_60']:.2f}
- MA240: {latest['SMA_240']:.2f}
- 收盤價: {latest['close']:.2f}

趨勢判斷（只回答）：強多頭 / 多頭 / 盤整 / 空頭 / 強空頭"""
        
        trend_direction = self.ask_model(prompt_1)
        
        # ==== 問題2：動能強度（帶趨勢判斷）====
        prompt_2 = f"""趨勢為「{trend_direction}」。

動能指標：
- RSI(14): {latest['RSI_14']:.1f}
- MACD: {latest['MACD']:.2f}
- CCI: {latest['CCI']:.1f}

動能是否過熱/過冷？只回答：嚴重超買 / 超買 / 中性 / 超賣 / 嚴重超賣"""
        
        momentum_status = self.ask_model(prompt_2)
        
        # ==== 問題3：支撐壓力位置 ====
        prompt_3 = f"""趨勢「{trend_direction}」，動能「{momentum_status}」。

位置數據：
- 當前價: {latest['close']:.2f}
- 布林上軌: {latest['BB_upper']:.2f}
- 布林中軌: {latest['BB_middle']:.2f}
- 布林下軌: {latest['BB_lower']:.2f}
- 距MA60: {(latest['close']/latest['SMA_60']-1)*100:+.2f}%

當前位置判斷？只回答：壓力區 / 接近壓力 / 中性區 / 接近支撐 / 支撐區"""
        
        position_status = self.ask_model(prompt_3)
        
        # ==== 問題4：技術型態識別驗證 ====
        patterns_found = []
        if latest['DOJI'] != 0:
            patterns_found.append('十字線')
        if latest['HAMMER'] != 0:
            patterns_found.append('錘子線')
        if latest['ENGULFING'] != 0:
            patterns_found.append('吞噬型態')
            
        prompt_4 = f"""技術面摘要：
- 趨勢: {trend_direction}
- 動能: {momentum_status}
- 位置: {position_status}
- 型態: {', '.join(patterns_found) if patterns_found else '無明顯型態'}

這些訊號是否一致？只回答：強烈一致 / 多數一致 / 部分矛盾 / 嚴重矛盾"""
        
        signal_consistency = self.ask_model(prompt_4)
        
        # ==== 問題5：技術面風險評估 ====
        prompt_5 = f"""綜合技術面風險。

當前狀況：
- 趨勢: {trend_direction}
- 動能: {momentum_status}
- 位置: {position_status}
- 訊號一致性: {signal_consistency}
- ATR波動率: {latest['ATR']:.2f}

技術面風險等級（只回答）：極高風險 / 高風險 / 中等風險 / 低風險 / 極低風險"""
        
        technical_risk = self.ask_model(prompt_5)
        
        return {
            'trend_direction': trend_direction,
            'momentum_status': momentum_status,
            'position_status': position_status,
            'signal_consistency': signal_consistency,
            'technical_risk': technical_risk,
            'indicators_raw': latest.to_dict()  # 保留原始數據供驗證
        }
```


### 第四層：跨維度整合決策（8B模型）

```python
class FinalDecisionMaker:
    """最終決策整合器（使用8B-14B模型）"""
    
    def __init__(self):
        self.model = "deepseek-r1:8b"  # 推理能力強的模型
    
    def make_final_decision(self, date, symbol, all_analysis):
        """整合所有分析，做出最終決策（多次驗證）"""
        
        # ==== 決策問題1：各維度權重評估 ====
        prompt_1 = f"""評估各分析維度的可信度。

籌碼面：
- 外資: {all_analysis['chip']['foreign_attitude']}
- 投信: {all_analysis['chip']['trust_attitude']}
- 籌碼穩定度: {all_analysis['chip']['chip_stability_score']}/100
- 散戶情緒: {all_analysis['chip']['retail_sentiment']}

技術面：
- 趨勢: {all_analysis['technical']['trend_direction']}
- 動能: {all_analysis['technical']['momentum_status']}
- 訊號一致性: {all_analysis['technical']['signal_consistency']}
- 技術風險: {all_analysis['technical']['technical_risk']}

基本面：
- 月營收YoY: {all_analysis['fundamental']['revenue_yoy']:+.2%}
- 近期新聞: {all_analysis['news']['sentiment']}

給出權重分配（總和100%）：
- 籌碼面權重: ___%
- 技術面權重: ___%
- 基本面權重: ___%"""
        
        weights = self.parse_weights(self.ask_model(prompt_1))
        
        # ==== 決策問題2：矛盾點識別 ====
        prompt_2 = f"""識別分析中的矛盾。

觀察：
- 外資「{all_analysis['chip']['foreign_attitude']}」但技術面「{all_analysis['technical']['trend_direction']}」
- 散戶「{all_analysis['chip']['retail_sentiment']}」與主力「{all_analysis['chip']['retail_vs_main']}」
- 技術訊號「{all_analysis['technical']['signal_consistency']}」

是否存在重大矛盾？說明原因（不超過30字）："""
        
        contradiction = self.ask_model(prompt_2)
        
        # ==== 決策問題3：交易方向 ====
        prompt_3 = f"""決定交易方向。

權重分配：
- 籌碼面 {weights['chip']}%：{all_analysis['chip']['chip_stability_score']}/100分
- 技術面 {weights['technical']}%：風險{all_analysis['technical']['technical_risk']}
- 基本面 {weights['fundamental']}%：營收{all_analysis['fundamental']['revenue_yoy']:+.2%}

矛盾點：{contradiction}

交易方向（只回答）：強力買進 / 買進 / 觀望 / 減碼 / 賣出"""
        
        action = self.ask_model(prompt_3)
        
        # ==== 決策問題4：倉位大小（帶方向判斷）====
        prompt_4 = f"""決定「{action}」的倉位。

風險評估：
- 籌碼穩定度: {all_analysis['chip']['chip_stability_score']}/100
- 技術風險: {all_analysis['technical']['technical_risk']}
- 主力成本: {all_analysis['chip']['cost_advantage']}
- 訊號一致性: {all_analysis['technical']['signal_consistency']}

當前持倉: {self.get_current_position(symbol)*100:.0f}%
可用資金: {self.get_available_cash()*100:.0f}%

建議倉位百分比（小數，例如0.15）："""
        
        position_size = float(self.ask_model(prompt_4) or "0")
        
        # ==== 決策問題5：止損止盈設定 ====
        prompt_5 = f"""設定風控參數。

決策：{action}，倉位{position_size*100:.0f}%

風險數據：
- ATR波動率: {all_analysis['technical']['indicators_raw']['ATR']:.2f}
- 支撐距離: {all_analysis['technical']['position_status']}
- 籌碼穩定: {all_analysis['chip']['chip_stability_score']}/100

建議止損%（小數）：
建議止盈%（小數）："""
        
        stop_loss, take_profit = self.parse_stops(self.ask_model(prompt_5))
        
        # ==== 決策問題6：決策信心度 ====
        prompt_6 = f"""評估整體決策信心。

決策鏈：
1. 方向: {action}
2. 倉位: {position_size*100:.0f}%
3. 止損: {stop_loss*100:.1f}%
4. 止盈: {take_profit*100:.1f}%
5. 矛盾點: {contradiction}

信心度（只回答）：A（極高）/ B（高）/ C（中）/ D（低）"""
        
        confidence = self.ask_model(prompt_6)
        
        # ==== 決策問題7：決策理由摘要 ====
        prompt_7 = f"""總結決策邏輯。

核心依據：
- 籌碼面: 外資{all_analysis['chip']['foreign_attitude']}、穩定度{all_analysis['chip']['chip_stability_score']}
- 技術面: 趨勢{all_analysis['technical']['trend_direction']}、風險{all_analysis['technical']['technical_risk']}
- 基本面: 營收{all_analysis['fundamental']['revenue_yoy']:+.2%}

用一句話說明為何「{action}」（不超過40字）："""
        
        reasoning = self.ask_model(prompt_7)
        
        return {
            'action': action,
            'position_size': position_size,
            'stop_loss': stop_loss,
            'take_profit': take_profit,
            'confidence': confidence,
            'reasoning': reasoning,
            'weights': weights,
            'contradiction': contradiction
        }
```
